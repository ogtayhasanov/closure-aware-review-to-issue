def canonical_stem(path: Path) -> str:
    return re.sub(r"\s*\(\d+\)$", "", path.stem.lower())


def norm_id(value: object) -> str:
    if pd.isna(value):
        return ""
    if isinstance(value, (int, np.integer)):
        return str(int(value))
    if isinstance(value, (float, np.floating)) and float(value).is_integer():
        return str(int(value))
    text = str(value).strip()
    if re.fullmatch(r"-?\d+\.0", text):
        return text[:-2]
    return text


def clean_scalar(value: object) -> str:
    return "" if pd.isna(value) else str(value).strip()


def dataset_name(filename: str) -> str:
    return (
        filename.replace("_issues_matching.csv", "")
        .replace("_matching.csv", "")
        .replace("_", " ")
        .title()
    )


def find_evaluation_file(filename: str) -> Path:
    wanted = Path(filename).stem.lower()
    matches = [
        path
        for path in INPUT_ROOT.rglob("*.csv")
        if canonical_stem(path) == wanted
    ]
    if not matches:
        raise FileNotFoundError(
            f"Missing {filename}. Attach the datasets containing the three "
            "Dolphin, Jitsi, and PPSSPP matching CSV files."
        )
    matches.sort(
        key=lambda path: (
            "matchessss" not in str(path).lower(),
            path.name.lower() != filename.lower(),
            len(str(path)),
        )
    )
    return matches[0]


def is_model_directory(path: Path) -> bool:
    return (
        (path / "config.json").exists()
        and (
            (path / "model.safetensors").exists()
            or (path / "pytorch_model.bin").exists()
        )
        and (path / "tokenizer_config.json").exists()
    )


def parse_old35_state(state_path: Path, expected_seed: int):
    try:
        state = json.loads(state_path.read_text(encoding="utf-8"))
    except Exception:
        return None

    # Reject the discarded query-disjoint structure. The required checkpoints
    # use seed/step/val_loss, not train_seed/epoch/validation_loss.
    if "train_seed" in state or "epoch" in state or "validation_loss" in state:
        return None
    try:
        seed = int(state["seed"])
        step = int(state["step"])
        val_loss = float(state["val_loss"])
        model_name = str(state["model_name"])
        max_length = int(state["max_length"])
        batch_size = int(state["batch_size"])
        grad_accum = int(state["grad_accum"])
        epochs = int(state["epochs"])
        temperature = float(state["temperature"])
        splade_lambda = float(state["splade_lambda"])
        learning_rate = float(state["lr"])
        weight_decay = float(state["weight_decay"])
    except (KeyError, TypeError, ValueError):
        return None

    if seed != expected_seed or not math.isfinite(val_loss):
        return None
    if step < 200 or step > MAX_SELECTION_STEP or step % 200 != 0:
        return None
    expected = {
        "model_name": BASE_SPLADE,
        "max_length": 128,
        "batch_size": 2,
        "grad_accum": 16,
        "epochs": 3,
    }
    actual = {
        "model_name": model_name,
        "max_length": max_length,
        "batch_size": batch_size,
        "grad_accum": grad_accum,
        "epochs": epochs,
    }
    if actual != expected:
        return None
    numerical = (
        (temperature, 1.0),
        (splade_lambda, 1e-4),
        (learning_rate, 2e-5),
        (weight_decay, 0.01),
    )
    if any(
        not math.isclose(got, wanted, rel_tol=0.0, abs_tol=1e-12)
        for got, wanted in numerical
    ):
        return None
    checkpoint = state_path.parent
    if not is_model_directory(checkpoint):
        return None
    return {
        "path": checkpoint,
        "step": step,
        "val_loss": val_loss,
    }


def find_checkpoint(seed: int) -> Path:
    candidates = []
    for state_path in INPUT_ROOT.rglob("training_state.json"):
        candidate = parse_old35_state(state_path, seed)
        if candidate is not None:
            candidates.append(candidate)
    if not candidates:
        discovered = []
        for state_path in INPUT_ROOT.rglob("training_state.json"):
            try:
                state = json.loads(state_path.read_text(encoding="utf-8"))
            except Exception:
                state = {"unreadable": True}
            discovered.append(
                {
                    "path": str(state_path),
                    "seed_fields": {
                        key: state.get(key)
                        for key in ("seed", "train_seed", "step", "epoch")
                        if key in state
                    },
                }
            )
        raise FileNotFoundError(
            f"No valid OLD-35 checkpoint found for seed {seed}. "
            f"Discovered states: {discovered}"
        )
    candidates.sort(
        key=lambda item: (
            item["val_loss"],
            "best" not in str(item["path"]).lower(),
            len(str(item["path"])),
        )
    )
    selected = candidates[0]
    checkpoint_audit_rows.append(
        {
            "seed": seed,
            "checkpoint": str(selected["path"]),
            "best_step": selected["step"],
            "validation_loss": selected["val_loss"],
        }
    )
    print(
        f"seed={seed} checkpoint={selected['path']} "
        f"step={selected['step']} val_loss={selected['val_loss']:.6f}",
        flush=True,
    )
    return selected["path"]


def prepare_evaluation_data(frame: pd.DataFrame):
    required = {"issue_id", "issue_text", "issue_body", "review_id", "review_text"}
    missing = required - set(frame.columns)
    if missing:
        raise KeyError(f"Evaluation file is missing columns: {sorted(missing)}")

    issues = (
        frame[["issue_id", "issue_text", "issue_body"]]
        .drop_duplicates("issue_id")
        .reset_index(drop=True)
    )
    issue_ids = issues["issue_id"].map(norm_id).tolist()
    issue_texts = [
        f"{clean_scalar(row.issue_text)} "
        f"{clean_scalar(row.issue_body)[:500]}".strip()
        for row in issues.itertuples(index=False)
    ]
    issue_to_index = {
        issue_id: index for index, issue_id in enumerate(issue_ids)
    }

    reviews = (
        frame[["review_id", "review_text"]]
        .drop_duplicates("review_id")
        .reset_index(drop=True)
    )
    review_ids = reviews["review_id"].map(norm_id).tolist()
    review_texts = reviews["review_text"].map(clean_scalar).tolist()
    review_to_index = {
        review_id: index for index, review_id in enumerate(review_ids)
    }

    ground_truth: Dict[int, Set[int]] = {}
    for row in frame.itertuples(index=False):
        review_index = review_to_index[norm_id(row.review_id)]
        issue_index = issue_to_index[norm_id(row.issue_id)]
        ground_truth.setdefault(review_index, set()).add(issue_index)

    return {
        "review_ids": review_ids,
        "review_texts": review_texts,
        "issue_ids": issue_ids,
        "issue_texts": issue_texts,
        "ground_truth": ground_truth,
    }


def scores_to_rankings(scores: np.ndarray):
    return [
        np.argsort(scores[index], kind="stable")[::-1].tolist()
        for index in range(scores.shape[0])
    ]


def query_metrics(ranking: Sequence[int], relevant: Set[int]):
    relevant_ranks = [
        rank
        for rank, issue_index in enumerate(ranking, start=1)
        if issue_index in relevant
    ]
    best_rank = min(relevant_ranks) if relevant_ranks else 0
    dcg10 = sum(
        1.0 / np.log2(rank + 1)
        for rank, issue_index in enumerate(ranking[:10], start=1)
        if issue_index in relevant
    )
    ideal_count = min(len(relevant), 10)
    idcg10 = sum(
        1.0 / np.log2(rank + 1)
        for rank in range(1, ideal_count + 1)
    )
    return {
        "best_relevant_rank": best_rank,
        "hit_at_1": int(bool(best_rank and best_rank <= 1)),
        "hit_at_5": int(bool(best_rank and best_rank <= 5)),
        "hit_at_10": int(bool(best_rank and best_rank <= 10)),
        "reciprocal_rank": 1.0 / best_rank if best_rank else 0.0,
        "ndcg_at_10": dcg10 / idcg10 if idcg10 else 0.0,
    }


def record_rankings(
    method: str,
    stage: str,
    depth: int,
    seed,
    dataset: str,
    data,
    rankings,
):
    for review_index, ranking in enumerate(rankings):
        relevant = data["ground_truth"][review_index]
        metrics = query_metrics(ranking, relevant)
        all_query_rows.append(
            {
                "method": method,
                "stage": stage,
                "ce_depth": depth,
                "seed": seed,
                "dataset": dataset,
                "review_id": data["review_ids"][review_index],
                "query_key": f"{dataset}::{data['review_ids'][review_index]}",
                "relevant_issue_ids": "|".join(
                    data["issue_ids"][index] for index in sorted(relevant)
                ),
                **metrics,
            }
        )


def bm25_rankings(review_texts, issue_texts):
    model = BM25Okapi([text.lower().split() for text in issue_texts])
    scores = np.vstack(
        [model.get_scores(text.lower().split()) for text in review_texts]
    )
    return scores_to_rankings(scores)


def dense_rankings(
    model: SentenceTransformer,
    review_texts,
    issue_texts,
    query_prefix="",
    document_prefix="",
):
    queries = [query_prefix + text for text in review_texts]
    documents = [document_prefix + text for text in issue_texts]
    query_vectors = model.encode(
        queries,
        batch_size=32,
        show_progress_bar=False,
        convert_to_tensor=True,
        normalize_embeddings=True,
    )
    document_vectors = model.encode(
        documents,
        batch_size=32,
        show_progress_bar=False,
        convert_to_tensor=True,
        normalize_embeddings=True,
    )
    scores = torch.mm(query_vectors, document_vectors.t()).cpu().numpy()
    del query_vectors, document_vectors
    return scores_to_rankings(scores)


class SpladeEncoder(nn.Module):
    def __init__(self, checkpoint):
        super().__init__()
        self.model = AutoModelForMaskedLM.from_pretrained(checkpoint)

    def forward(self, tokens, device):
        moved = {
            key: value.to(device, non_blocking=True)
            for key, value in tokens.items()
        }
        logits = self.model(**moved).logits
        values = (
            torch.log1p(torch.relu(logits))
            * moved["attention_mask"].unsqueeze(-1)
        )
        return torch.max(values, dim=1).values


def splade_rankings(checkpoint, review_texts, issue_texts, device):
    tokenizer = AutoTokenizer.from_pretrained(checkpoint)
    model = SpladeEncoder(checkpoint).to(device)
    model.eval()

    @torch.inference_mode()
    def encode(texts, batch_size=8):
        vectors = []
        for start in range(0, len(texts), batch_size):
            tokens = tokenizer(
                list(texts[start : start + batch_size]),
                max_length=128,
                truncation=True,
                padding=True,
                return_tensors="pt",
            )
            vectors.append(model(tokens, device).cpu())
        return torch.cat(vectors, dim=0)

    review_vectors = encode(review_texts)
    issue_vectors = encode(issue_texts)
    scores = torch.mm(review_vectors, issue_vectors.t()).numpy()
    rankings = scores_to_rankings(scores)
    del model, tokenizer, review_vectors, issue_vectors, scores
    gc.collect()
    torch.cuda.empty_cache()
    return rankings


def rerank_at_depths(
    cross_encoder,
    review_texts,
    issue_texts,
    first_stage_rankings,
    depths,
):
    maximum_depth = max(depths)
    pairs = []
    offsets = []
    for review_index, ranking in enumerate(first_stage_rankings):
        candidates = ranking[:maximum_depth]
        start = len(pairs)
        pairs.extend(
            (review_texts[review_index], issue_texts[issue_index])
            for issue_index in candidates
        )
        offsets.append((start, len(candidates)))
    scores = np.asarray(
        cross_encoder.predict(
            pairs,
            batch_size=64,
            show_progress_bar=False,
        )
    ).reshape(-1)

    output = {depth: [] for depth in depths}
    for review_index, initial in enumerate(first_stage_rankings):
        start, count = offsets[review_index]
        candidate_scores = scores[start : start + count]
        score_by_issue = {
            issue_index: float(candidate_scores[position])
            for position, issue_index in enumerate(initial[:count])
        }
        for depth in depths:
            candidates = initial[:depth]
            reranked = sorted(
                candidates,
                key=lambda issue_index: score_by_issue[issue_index],
                reverse=True,
            )
            candidate_set = set(candidates)
            remainder = [
                issue_index
                for issue_index in initial
                if issue_index not in candidate_set
            ]
            output[depth].append(reranked + remainder)
    return output


def aggregate_query_rows(per_query: pd.DataFrame):
    rows = []
    keys = ["method", "stage", "ce_depth", "seed"]
    for key, subset in per_query.groupby(keys, dropna=False):
        method, stage, depth, seed = key
        for dataset, dataset_rows in subset.groupby("dataset"):
            row = {
                "method": method,
                "stage": stage,
                "ce_depth": int(depth),
                "seed": seed,
                "aggregation": "Per application",
                "dataset": dataset,
                "queries": len(dataset_rows),
            }
            for metric, label in METRIC_LABELS.items():
                row[label] = float(dataset_rows[metric].mean())
            rows.append(row)

        pooled = {
            "method": method,
            "stage": stage,
            "ce_depth": int(depth),
            "seed": seed,
            "aggregation": "Pooled",
            "dataset": "Overall",
            "queries": len(subset),
        }
        for metric, label in METRIC_LABELS.items():
            pooled[label] = float(subset[metric].mean())
        rows.append(pooled)

        application_means = subset.groupby("dataset")[list(METRIC_COLUMNS)].mean()
        macro = {
            "method": method,
            "stage": stage,
            "ce_depth": int(depth),
            "seed": seed,
            "aggregation": "Macro",
            "dataset": "Overall",
            "queries": len(subset),
        }
        for metric, label in METRIC_LABELS.items():
            macro[label] = float(application_means[metric].mean())
        rows.append(macro)
    return pd.DataFrame(rows)


def seed_statistics(summary: pd.DataFrame):
    seeded = summary[summary["seed"].notna()].copy()
    rows = []
    t_critical_95_df4 = 2.7764451051977987
    grouping = ["method", "stage", "ce_depth", "aggregation", "dataset"]
    for key, subset in seeded.groupby(grouping):
        if len(subset) != len(SEEDS):
            raise RuntimeError(
                f"Expected {len(SEEDS)} seeds for {key}, found {len(subset)}"
            )
        row = dict(zip(grouping, key))
        row["seeds"] = len(subset)
        row["queries_per_seed"] = int(subset["queries"].iloc[0])
        for label in METRIC_LABELS.values():
            values = subset[label].astype(float)
            mean = float(values.mean())
            standard_deviation = float(values.std(ddof=1))
            half_width = (
                t_critical_95_df4
                * standard_deviation
                / math.sqrt(len(values))
            )
            row[f"{label}_mean"] = mean
            row[f"{label}_std"] = standard_deviation
            row[f"{label}_ci95_low"] = max(0.0, mean - half_width)
            row[f"{label}_ci95_high"] = min(1.0, mean + half_width)
        rows.append(row)
    return pd.DataFrame(rows)


def bootstrap_fixed_baselines(per_query: pd.DataFrame, repetitions=5000):
    fixed = per_query[per_query["seed"].isna()].copy()
    rng = np.random.default_rng(20260728)
    rows = []
    grouping = ["method", "stage", "ce_depth"]
    for key, subset in fixed.groupby(grouping):
        method, stage, depth = key
        datasets = sorted(subset["dataset"].unique())
        by_dataset = {
            dataset: subset[subset["dataset"] == dataset].reset_index(drop=True)
            for dataset in datasets
        }
        for aggregation in ("Pooled", "Macro"):
            row = {
                "method": method,
                "stage": stage,
                "ce_depth": int(depth),
                "aggregation": aggregation,
                "bootstrap_repetitions": repetitions,
            }
            for metric, label in METRIC_LABELS.items():
                draws = []
                for _ in range(repetitions):
                    sampled_means = []
                    sampled_values = []
                    for dataset in datasets:
                        values = by_dataset[dataset][metric].to_numpy(float)
                        indices = rng.integers(0, len(values), len(values))
                        sample = values[indices]
                        sampled_means.append(float(sample.mean()))
                        sampled_values.extend(sample.tolist())
                    if aggregation == "Pooled":
                        draws.append(float(np.mean(sampled_values)))
                    else:
                        draws.append(float(np.mean(sampled_means)))
                point = (
                    float(subset[metric].mean())
                    if aggregation == "Pooled"
                    else float(
                        subset.groupby("dataset")[metric].mean().mean()
                    )
                )
                row[f"{label}_estimate"] = point
                row[f"{label}_ci95_low"] = float(np.quantile(draws, 0.025))
                row[f"{label}_ci95_high"] = float(np.quantile(draws, 0.975))
            rows.append(row)
    return pd.DataFrame(rows)


def holm_adjust(p_values):
    p_values = np.asarray(p_values, dtype=float)
    order = np.argsort(p_values)
    adjusted = np.empty(len(p_values), dtype=float)
    running_max = 0.0
    for rank, index in enumerate(order):
        candidate = min(1.0, (len(p_values) - rank) * p_values[index])
        running_max = max(running_max, candidate)
        adjusted[index] = running_max
    return adjusted


def paired_tests(per_query: pd.DataFrame):
    rows = []
    splade_single = per_query[
        (per_query["method"] == "SPLADE-SE")
        & (per_query["stage"] == "single")
    ]
    splade_single = (
        splade_single.groupby("query_key")[list(METRIC_COLUMNS)].mean()
    )

    # RQ1: five-seed mean contribution per query versus fixed baselines.
    for baseline in ("BM25", "MPNet", "BGE-Base", "E5-Base", "SPLADE-ZS"):
        baseline_rows = per_query[
            (per_query["method"] == baseline)
            & (per_query["stage"] == "single")
        ].set_index("query_key")
        for metric, label in METRIC_LABELS.items():
            left = splade_single[metric].sort_index()
            right = baseline_rows.loc[left.index, metric].astype(float)
            differences = left.to_numpy() - right.to_numpy()
            p_value = (
                1.0
                if np.allclose(differences, 0.0)
                else float(
                    wilcoxon(
                        differences,
                        zero_method="pratt",
                        alternative="two-sided",
                    ).pvalue
                )
            )
            rows.append(
                {
                    "research_question": "RQ1",
                    "comparison": f"SPLADE-SE vs {baseline}",
                    "ce_depth": 0,
                    "metric": label,
                    "mean_paired_difference": float(differences.mean()),
                    "pairs": len(differences),
                    "test": "paired Wilcoxon over query contributions",
                    "p_raw": p_value,
                }
            )

    # RQ2: each matched CE depth versus the same five first-stage checkpoints.
    for depth in CE_DEPTHS:
        ce_rows = per_query[
            (per_query["method"] == "SPLADE-SE")
            & (per_query["stage"] == "cross-encoder")
            & (per_query["ce_depth"] == depth)
        ]
        ce_mean = ce_rows.groupby("query_key")[list(METRIC_COLUMNS)].mean()
        for metric, label in METRIC_LABELS.items():
            left = ce_mean.loc[splade_single.index, metric].to_numpy()
            right = splade_single[metric].to_numpy()
            differences = left - right
            p_value = (
                1.0
                if np.allclose(differences, 0.0)
                else float(
                    wilcoxon(
                        differences,
                        zero_method="pratt",
                        alternative="two-sided",
                    ).pvalue
                )
            )
            rows.append(
                {
                    "research_question": "RQ2",
                    "comparison": f"SPLADE-SE+CE@{depth} vs SPLADE-SE",
                    "ce_depth": depth,
                    "metric": label,
                    "mean_paired_difference": float(differences.mean()),
                    "pairs": len(differences),
                    "test": "paired Wilcoxon over query contributions",
                    "p_raw": p_value,
                }
            )

    output = pd.DataFrame(rows)
    output["p_holm_within_rq"] = np.nan
    for research_question, indices in output.groupby("research_question").groups.items():
        output.loc[indices, "p_holm_within_rq"] = holm_adjust(
            output.loc[indices, "p_raw"].to_numpy()
        )
    return output

    print(
        f"{name}: {len(datasets[name]['review_ids'])} reviews / "
        f"{len(datasets[name]['issue_ids'])} issues from {path}",
        flush=True,
    )
if sum(len(data["review_ids"]) for data in datasets.values()) != 102:
    raise RuntimeError("Expected exactly 102 unique evaluation reviews.")

rankings_cache = {}

print("\n1/4 Fixed single-stage baselines", flush=True)
for name, data in datasets.items():
    rankings = bm25_rankings(data["review_texts"], data["issue_texts"])
    rankings_cache[("BM25", None, name)] = rankings
    record_rankings("BM25", "single", 0, None, name, data, rankings)

dense_configs = (
    ("MPNet", "all-mpnet-base-v2", "", ""),
    ("BGE-Base", "BAAI/bge-base-en-v1.5", "", ""),
    ("E5-Base", "intfloat/e5-base-v2", "query: ", "passage: "),
)
for method, model_name, query_prefix, document_prefix in dense_configs:
    print(f"  loading {method}: {model_name}", flush=True)
    dense_model = SentenceTransformer(model_name, device=str(device))
    for name, data in datasets.items():
        rankings = dense_rankings(
            dense_model,
            data["review_texts"],
            data["issue_texts"],
            query_prefix,
            document_prefix,
        )
        rankings_cache[(method, None, name)] = rankings
        record_rankings(method, "single", 0, None, name, data, rankings)
    del dense_model
    gc.collect()
    torch.cuda.empty_cache()

print(f"  loading SPLADE-ZS: {BASE_SPLADE}", flush=True)
for name, data in datasets.items():
    rankings = splade_rankings(
        BASE_SPLADE,
        data["review_texts"],
        data["issue_texts"],
        device,
    )
    rankings_cache[("SPLADE-ZS", None, name)] = rankings
    record_rankings("SPLADE-ZS", "single", 0, None, name, data, rankings)

print("\n2/4 Five matched SPLADE-SE checkpoints", flush=True)
for seed in SEEDS:
    checkpoint = checkpoints[seed]
    for name, data in datasets.items():
        print(f"  seed={seed} dataset={name}", flush=True)
        rankings = splade_rankings(
            checkpoint,
            data["review_texts"],
            data["issue_texts"],
            device,
        )
        rankings_cache[("SPLADE-SE", seed, name)] = rankings
        record_rankings(

            "single",
            0,
            seed,
            name,
            data,
            rankings,
        )

print("\n3/4 Matched cross-encoder reranking at top-k = 5, 10, 20", flush=True)
cross_encoder = CrossEncoder(CE_MODEL, device=str(device))
ce_jobs = [
    ("BGE-Base", None),
    ("E5-Base", None),
    *[("SPLADE-SE", seed) for seed in SEEDS],
]
for method, seed in ce_jobs:
    for name, data in datasets.items():
        print(f"  {method} seed={seed} dataset={name}", flush=True)
        first_stage = rankings_cache[(method, seed, name)]
        reranked = rerank_at_depths(
            cross_encoder,
            data["review_texts"],
            data["issue_texts"],
            first_stage,
            CE_DEPTHS,
        )
        for depth, rankings in reranked.items():
            record_rankings(
                method,
                "cross-encoder",
                depth,
                seed,
                name,
                data,
                rankings,
            )
del cross_encoder
